-:: Lecture note ::-
- Data insertion should follow the column order & respective datatype.
- for date follow  this order: "year-month-day".


*** Commands ***

-- insert row into a table.
INSERT INTO table_name
VALUES (value1, value2, value3, ...);


-- insert multiple rows into a table.
INSERT INTO table_name
VALUES 
(value1, value2, value3, ...),
(value1, value2, value3, ...),
.
.
(value1, value2, value3, ...);


-- insert row or rows, having data for only selected columns. In those rows, data for the unselected columns will be null.
INSERT INTO table_name (column1, column2, column3, ...)
VALUES (value1, value2, value3, ...);
